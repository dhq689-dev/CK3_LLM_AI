# Smoke-test on_actions.
#
# IMPORTANT LESSON:
# An on_action's `effect` is a SINGLE block and does NOT merge across files --
# the base game's version wins and ours is discarded. We must instead APPEND to
# the LIST fields (`on_actions` / `events`), which DO merge.
#
# So we add our own on_action to the list, and put our effect inside it.

on_game_start = {
	on_actions = { ck3llm_on_game_start }
}

yearly_global_pulse = {
	on_actions = { ck3llm_yearly_pulse }
}

# --- our custom on_actions -----------------------------------------------

ck3llm_on_game_start = {
	effect = {
		ck3llm_smoke_test_player = yes
	}
}

ck3llm_yearly_pulse = {
	effect = {
		ck3llm_smoke_test_title = yes
	}
}
